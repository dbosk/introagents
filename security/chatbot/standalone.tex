% The support chatbot's literate program typeset on its own.  The body,
% chatbot.tex, is woven from chatbot.nw as a chapter without a preamble,
% so that the lecture notes can include it as an appendix; this driver
% gives it a title, a bibliography and the indices.  It loads only
% packages that the lecture's preamble (../preamble.tex) also loads,
% plus noweb, so that a body that builds here builds there too.

\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[swedish,british]{babel}
\usepackage{booktabs}
\usepackage{import}
\usepackage{graphicx}
\usepackage{tikz}
\usetikzlibrary{positioning,arrows.meta,fit}
\usepackage[strict]{csquotes}
\usepackage{minted}
\usepackage[natbib,style=verbose,citestyle=verbose,singletitle=false,
            maxbibnames=99]{biblatex}

\usepackage[colorlinks]{hyperref}
\usepackage[capitalise]{cleveref}

\usepackage[minted]{noweb}
